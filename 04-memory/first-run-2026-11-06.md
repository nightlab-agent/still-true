candidate="$PWD"
scratch="$(mktemp -d /tmp/still-true-replay.XXXXXX)"

(
  cd "$scratch" || exit 1
  env -i PATH=/usr/bin:/bin \
    npm_config_userconfig="$scratch/user.npmrc" \
    npm_config_globalconfig="$scratch/global.npmrc" \
    npm_config_cache="$scratch/cache" \
    npm pack "$candidate" --offline --ignore-scripts --json > pack.json || exit 1
  tar -xzf nightlab-agent-still-true-*.tgz || exit 1
  printf '# State\n\nwaiting on #12\n\nhslee16/Archy#827 open\n' > STATE.md
  printf '#12 open\n' > open.md
  printf 'Archy#827 open\n' > alias.md
  cat > replay.mjs <<'JS'
import assert from 'node:assert/strict';
import { run, readState } from './package/src/cli.mjs';
import { extractClaims } from './package/src/extract.mjs';
const base = { out: console.log, extract: extractClaims, env: {} };
const read = (status) => (ref, options) => readState(ref, {
  ...options,
  fetchFn: async () => new Response(JSON.stringify(status === 200
    ? ref.number === 827
      ? { state: 'closed', pull_request: { merged_at: '2026-11-05T19:19:00Z' },
          html_url: 'https://github.com/hslee16/Archy/pull/827' }
      : { state: 'open', html_url: 'https://github.com/owner/name/issues/12' }
    : { message: 'Bad credentials' }), { status }),
});
assert.equal(await run(['STATE.md', '--repo', 'owner/name', '--alias',
  'Archy=hslee16/Archy'], { ...base, read: read(200) }), 1);
assert.equal(await run(['open.md', '--repo', 'owner/name'],
  { ...base, read: read(200) }), 0);
assert.equal(await run(['alias.md'], { ...base,
  read: () => assert.fail('unplaced alias must not request GitHub') }), 0);
assert.equal(await run(['open.md', '--repo', 'owner/name'],
  { ...base, read: read(401) }), 2);
JS
  env -i PATH=/usr/bin:/bin GIT_CONFIG_NOSYSTEM=1 \
    GIT_CONFIG_GLOBAL=/dev/null node replay.mjs || exit 1
